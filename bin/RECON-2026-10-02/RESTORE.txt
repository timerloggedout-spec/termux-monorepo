import os
from pathlib import Path
HOME = Path.home()

# gpg-prime — primes agent with all keygrips, no TTL
(BIN / "gpg-prime").write_text('''#!/data/data/com.termux/files/usr/bin/bash
# @tool gpg-prime
# Preset the gpg-agent cache for every keygrip (primary + encryption subkey).
set -u
export GPG_TTY="$(tty 2>/dev/null || echo /dev/null)"
PF="$HOME/.gnupg/.2fa-pass"
[ -f "$PF" ] || { echo "gpg-prime: no $PF"; exit 1; }
PW="$(cat "$PF")"
KEYGRIPS=$(gpg --list-secret-keys --with-keygrip 2>/dev/null | awk '/Keygrip/{print $3}')
[ -z "$KEYGRIPS" ] && { echo "gpg-prime: no keygrips found"; exit 1; }
for kg in $KEYGRIPS; do
  printf 'PRESET_PASSPHRASE %s -1 %s\\n' "$kg" "$(printf '%s' "$PW" | xxd -p -c 200)" \\
    | gpg-connect-agent --quiet >/dev/null 2>&1 || true
done
echo "gpg-prime: primed $(echo "$KEYGRIPS" | wc -l) keygrip(s)"
''')
os.chmod(BIN / "gpg-prime", 0o755)

# gh2fa — print current TOTP, re-prime on failure
(BIN / "gh2fa").write_text('''#!/data/data/com.termux/files/usr/bin/bash
# @tool gh2fa
set -u
export GPG_TTY="$(tty 2>/dev/null || echo /dev/null)"
code="$(pass otp secret/github 2>/dev/null)" || code=""

  "$HOME/bin/gpg-prime" >/dev/null 2>&1 || true
  code="$(pass otp secret/github 2>/dev/null)" || code=""
fi
[ -n "$code" ] && printf '%s\\n' "$code" || { echo "gh2fa: failed" >&2; exit 1; }
''')

import sys
from pathlib import Path
HOME = Path.home()
sys.path.insert(0, str(HOME / "deepcli"))
from deepcli.core import get_token, get_history, fetch_sessions

tok = get_token()
sessions = fetch_sessions(tok)
print(f"scanning {len(sessions)} sessions for script keywords")

needles = ["gpg-prime", "gh-code", "gh2fa", "2fa-pass", "pass otp", "PRESET_PASSPHRASE", "gpg-connect-agent"]
found = []
for s in sessions:
    sid = s.get("id") or s.get("chat_session_id")
    if not sid: continue
    try:
        msgs = get_history(tok, sid, force_refresh=False)
    except Exception:
        continue
    n = 0
    for m in msgs or []:
        c = m.get("content") or ""
        if any(k in c for k in needles):
            n += 1
    if n:
        title = s.get("title") or s.get("name") or "?"
        print(f"  HIT {sid[:16]}  msgs={len(msgs)}  matches={n}  {title[:60]}")
        found.append((sid, n, title))

print(f"\ntotal: {len(found)}")
